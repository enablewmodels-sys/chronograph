cmake_minimum_required(VERSION 3.18)
project(chronograph_sdk VERSION 0.4.0 LANGUAGES CXX)
find_package(CURL REQUIRED)
find_package(nlohmann_json 3.12 REQUIRED)
add_library(chronograph INTERFACE)
add_library(chronograph::chronograph ALIAS chronograph)
target_compile_features(chronograph INTERFACE cxx_std_17)
target_include_directories(chronograph INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(chronograph INTERFACE CURL::libcurl nlohmann_json::nlohmann_json)
